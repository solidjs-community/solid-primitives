# REPL Test

A trial page for SolidBase's live REPL blocks. Edit the code and the preview updates.

## Single file

A signal and a derived memo, using only `solid-js`.

:::repl
```tsx
import { render } from "@solidjs/web";
import { createMemo, createSignal } from "solid-js";

function Counter() {
  const [count, setCount] = createSignal(1);
  const doubled = createMemo(() => count() * 2);

  return (
    <button type="button" onClick={() => setCount(c => c + 1)}>
      {count()} × 2 = {doubled()}
    </button>
  );
}

render(() => <Counter />, document.getElementById("app")!);
```

:::

## Using a primitive from this library

Imports from npm packages, here `@solid-primitives/timer`.

:::repl{layout=horizontal height="22rem"}
```tsx
import { render } from "@solidjs/web";
import { createSignal } from "solid-js";
import { createTimer } from "@solid-primitives/timer";

function Ticker() {
  const [ticks, setTicks] = createSignal(0);
  // Cleans itself up when the component is disposed.
  createTimer(() => setTicks(t => t + 1), 1000, setInterval);

  return <p>Ticks: {ticks()}</p>;
}

render(() => <Ticker />, document.getElementById("app")!);
```

  },

}

## Multiple files

Blocks after the first need a `title`, which becomes the file name.

:::repl
```tsx
import { render } from "@solidjs/web";
import { Greeting } from "./Greeting";

render(() => <Greeting name="Solid Primitives" />, document.getElementById("app")!);
```

```tsx title="Greeting.tsx"
export function Greeting(props: { name: string }) {
  return <h2>Hello {props.name}!</h2>;
}
```
