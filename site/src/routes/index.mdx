```tsx
import { render } from "@solidjs/web";
import { createSignal } from "solid-js";
import { createTimer } from "@solid-primitives/timer";
import { Clock } from "./Clock";
import { injectStyles } from "./styles";

const TOTAL = 30 * 60 * 1000; // 30 minutes

function App() {
  const [left, setLeft] = createSignal(TOTAL);
  const [running, setRunning] = createSignal(true); // starts automatically

  // The delay is reactive: `false` switches the timer off. It pauses when
  // `running` is false, stops itself at zero, and is cleaned up with the
  // component.
  createTimer(
    () => {
      const next = Math.max(0, left() - 1000);
      setLeft(next);
      if (next === 0) setRunning(false);
    },
    () => running() && 1000,
    setInterval,
  );

  return (
    <Clock
      left={left()}
      total={TOTAL}
      running={running()}
      onToggle={() => {
        if (left() === 0) setLeft(TOTAL);
        setRunning(r => !r);
      }}
      onReset={() => {
        setRunning(false);
        setLeft(TOTAL);
      }}
    />
  );
}

injectStyles();
render(() => <App />, document.getElementById("app")!);
```

```tsx title="Clock.tsx"
// Presentation only: a progress ring, the time left, and two buttons.
const R = 54;
const C = 2 * Math.PI * R;
const pad = (n: number) => String(n).padStart(2, "0");

function formatTime(ms: number) {
  const secs = Math.ceil(ms / 1000);
  return `${pad(Math.floor(secs / 60))}:${pad(secs % 60)}`;
}

export function Clock(props: {
  left: number;
  total: number;
  running: boolean;
  onToggle: () => void;
  onReset: () => void;
}) {
  const done = () => props.left === 0;

  return (
    <div class={done() ? "card done" : "card"}>
      <div class="dial">
        <svg viewBox="0 0 120 120">
          <circle class="track" cx="60" cy="60" r={R} />
          <circle
            class="bar"
            cx="60"
            cy="60"
            r={R}
            stroke-dasharray={C}
            stroke-dashoffset={C * (1 - props.left / props.total)}
          />
        </svg>
        <div class="time">{formatTime(props.left)}</div>
      </div>
      <div class="status">
        {done()
          ? "Time's up"
          : props.running
            ? "Counting down"
            : props.left < props.total
              ? "Paused"
              : "Ready"}
      </div>
      <div class="actions">
        <button class="primary" onClick={props.onToggle}>
          {done()
            ? "Restart"
            : props.running
              ? "Pause"
              : props.left < props.total
                ? "Resume"
                : "Start"}
        </button>
        <button onClick={props.onReset}>Reset</button>
      </div>
    </div>
  );
}
```
